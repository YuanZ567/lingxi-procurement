## 1. 收尾接口一：人工审批 `POST /api/task/{id}/approve`

### 1.1 后端（`src/api/task_api.py` 追加）

```python
@router.post("/{task_id}/approve")
async def approve_task(task_id: str):
    """人工审批口：manual 状态的单，人工确认后改成 completed（方案已生成，直接可执行）。
    注意：本路由带 /approve 子路径，但方法不同（POST），不会被 GET /{task_id} 吞掉"""
    reg = TASK_REGISTRY.get(task_id)                     # 先看内存登记表（任务可能还在本进程）
    with SessionLocal() as db:                           # 库是权威：无论如何都要改库
        task = db.get(ProcurementTask, task_id)          # 按主键查任务
        if not task:                                     # 查无此单
            raise HTTPException(status_code=404, detail={"code": 4004, "msg": "任务不存在"})
        if task.task_status != "manual":                 # 只有"待人工"的单才需要审批
            raise HTTPException(status_code=400, detail={"code": 4002, "msg": f"当前状态 {task.task_status} 无需人工审批"})
        task.task_status = "completed"                   # 人工拍板：方案可用，状态改完成
        db.commit()                                      # 提交落库
    if reg:                                              # 内存登记还在（本进程跑的单）
        reg["status"] = "completed"                      # 同步内存状态
        _push_event(task_id, {"event": "done", "data": "completed"})  # 给还在听的 SSE 补一条终态
    logger.info(f"[task {task_id}] 人工审批通过 → completed")
    return {"task_id": task_id, "status": "completed"}   # 前端拿它刷新界面
```

## 2. 收尾接口二：行情/供应商实时接口

### 2.1 后端：新建 `src/api/data_api.py`（🆕 整个文件）

```python
"""
data_api.py —— 看板数据接口（行情 + 供应商）
====================================================================
GET /api/market?commodity=螺纹钢&limit=260   单品种行情序列（最新在前可选）
GET /api/suppliers?region=天津市&category=螺纹钢   供应商档案（可按地区/品类筛）
让前端告别 public/*.json 静态文件，数据直接从 MySQL 来
"""

from fastapi import APIRouter, Query                  # 路由 + 查询参数校验
from sqlalchemy import desc                           # 倒序排序用
from src.db.session import SessionLocal               # 数据库会话
from src.db.models import CommodityPrice, Supplier    # 行情表 + 供应商表

router = APIRouter(prefix="/api", tags=["看板数据"])   # 共享前缀 /api

@router.get("/market")
async def get_market(
    commodity: str = Query("螺纹钢", description="品种中文名"),
    limit: int = Query(260, le=2603, description="返回最近多少个交易日")):
    """行情接口：从 commodity_price 表按品种倒序取最近 limit 条，正序返回给图表"""
    with SessionLocal() as db:                        # 开会话
        rows = (db.query(CommodityPrice)              # 查行情表
                  .filter(CommodityPrice.commodity_name == commodity)   # 只取该品种
                  .order_by(desc(CommodityPrice.record_date))           # 最新在前
                  .limit(limit).all())                # 截取条数
        rows.reverse()                                # 图表要"从旧到新"，把倒序翻回来
        return [{                                     # 组装成前端好用的格式
            "date": str(r.record_date),               # 交易日
            "close": float(r.price)                   # 收盘价（Decimal 转 float 才能进 JSON）
        } for r in rows]

@router.get("/suppliers")
async def get_suppliers(
    region: str = Query("", description="注册省份，空=全部"),
    category: str = Query("", description="主营品类，空=全部")):
    """供应商接口：从 supplier 表读真实工商档案+业务字段，支持地区/品类筛选"""
    with SessionLocal() as db:                        # 开会话
        q = db.query(Supplier)                        # 基础查询
        if region:                                    # 传了地区就加过滤
            q = q.filter(Supplier.province == region)
        if category:                                  # 传了品类就加过滤
            q = q.filter(Supplier.main_category == category)
        rows = q.all()                                # 执行查询
        return [{                                     # 组装返回（字段名与前端现用的一致，改造成本最低）
            "name": r.supplier_name,                  # 企业全名
            "credit": {"A": 95, "B": 75, "C": 55}.get(r.credit_status, 60),  # 等级转分数
            "premium": round((float(r.premium_rate) - 1) * 100, 2),           # 系数转溢价率%
            "region": r.province or "其他",           # 注册省份
            "category": r.main_category or "其他",    # 主营品类
        } for r in rows]
```

## 3. 收尾接口三：本月成本节约统计（真数据）

### 3.1 思路

每单"节约 = 预算 − 成交总额"。成交总额（`final_total_amount`）在谈判结果里，但没落库。所以分三步：
**① 表加列 `final_amount` → ② 任务终态时落库 → ③ stats 接口聚合本月完成单**。

### 3.5 统计接口（`task_api.py` 追加，放在 `GET /list` 前面）

```python
@router.get("/stats/month")
async def month_stats():
    """本月成本统计：本月完成的单数 + 总节约（预算−成交额）+ 平均预算占用率"""
    from datetime import datetime, date                # 取本月起点用
    month_start = date.today().replace(day=1)          # 本月1号
    with SessionLocal() as db:                         # 开会话
        rows = (db.query(ProcurementTask)              # 查任务表
                  .filter(ProcurementTask.task_status == "completed",          # 只算完成的单
                          ProcurementTask.create_time >= month_start)            # 只算本月的
                  .all())
        done = [t for t in rows if t.final_amount]     # 有成交额的单才参与节约计算
        saving = sum((float(t.budget_limit) - float(t.final_amount)) for t in done)  # 逐单节约再求和
        ratio = (sum(float(t.final_amount) / float(t.budget_limit) for t in done) / len(done) * 100) if done else 0
        return {                                       # 前端面板直接用的三个数
            "done_count": len(done),                   # 本月完成单数
            "total_saving": round(saving, 2),          # 总节约金额（元）
            "avg_budget_ratio": round(ratio, 1),       # 平均预算占用率（%）
        }
```

> ⚠️ 注意：`/stats/month` 也必须定义在 `GET /{task_id}` **之前**，否则 "stats" 会被当成 task_id（和 `/list` 同一个坑）。

```python
"""
acceptance_test.py —— P5 全流程验收脚本（对照 PRD §4.1~§4.5）
跑法：python acceptance_test.py
每通过一项打 ✅，失败打 ❌ 并说明；最后汇总通过率

        "purchase_quantity": 100, "expect_delivery_date": "2026-10-20", "budget_limit": 1000})
    tid_bad = r.json()["task_id"]                     # 超支单的 task_id

    import urllib.request                             # 直接查库最稳
```

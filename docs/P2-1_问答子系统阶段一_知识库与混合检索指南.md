## 0. 先用大白话搞懂：RAG 是什么

**问题**：LLM 只知道它训练时见过的公开资料，不知道"你们公司的采购制度"。
**RAG（检索增强生成）三步走**：

```
① 建库（本指南阶段一）：把制度文档切成小段 → 每段算出"语义指纹"（向量）→ 存进向量库
② 检索（本指南阶段一）：用户提问 → 把问题也算成向量 → 找出"语义指纹最像"的几段 + BM25关键词检索几段 → 两路结果融合
③ 生成（阶段二）：把找到的段落塞进 Prompt → LLM 照着段落回答 → 回答有据可查（溯源）
```

**为什么用"向量 + 关键词"两路检索（混合检索）**：
- 向量检索：懂"语义"——问"预算超了怎么办"能找到《预算管理办法》里的条款（哪怕没出现"预算超了"四个字）
- BM25 关键词检索：懂"精确词"——问"RB0 是什么"能精准命中写了"RB0"的段落
- 两路结果用 **RRF（倒数排名融合）** 合并：两边都排前面的段落最靠谱
- 这就是 TECH_SPEC §2.5 说的 Hybrid-RAG

## 2. `src/rag/vector_store.py`（新建，先建 `src/rag` 目录和 `__init__.py` 空文件）

> 前置：`pip show chromadb` 确认已装（requirements 里有）；`DASHSCOPE_API_KEY` 已配置

```python
# src/rag/vector_store.py
"""
向量库模块（RAG 的"语义检索员"）
====================================================================
干什么：把文本变成"语义指纹"（向量），存进 ChromaDB；
       查询时把用户问题也变成向量，找出语义最相近的文档片段。

向量是什么（大白话）：把一段文字压缩成一串数字（比如1024个），
意思相近的文字，这串数字就相近——"预算超支怎么办"和
《预算管理办法》里讲超支的段落，数字会离得很近。
"""

import chromadb                                   # ChromaDB：开源向量数据库，数据存本地文件夹
from langchain_openai import OpenAIEmbeddings     # 用百炼的 embedding 模型把文字变向量
from src.utils.config import get_settings         # 从配置中心拿 API Key

# 向量库的存放位置：项目根目录下的 chroma_db/ 文件夹（数据持久化在本地）
CHROMA_DIR = "chroma_db"
# 集合（collection）名字：可以理解为向量库里的"一张表"
COLLECTION_NAME = "lingxi_kb"

# 语义指纹生成器：调百炼的 text-embedding-v3 模型，把文字变成 1024 维向量
# check_embedding_ctx_length=False 必须加：告诉 langchain 别按 OpenAI 的方式预切分文本（百炼不认）
_embedder = OpenAIEmbeddings(
    model="text-embedding-v3",                    # 百炼的向量化模型（每次调用花几分钱）
    api_key=get_settings().DASHSCOPE_API_KEY,     # 密钥从配置中心读
    base_url=get_settings().LLM_BASE_URL,         # 接口地址同聊天模型（百炼兼容接口）
    dimensions=1024,                              # 输出向量的维度
    check_embedding_ctx_length=False,             # 关键参数：关闭 OpenAI 专用的预切分检查
)


def embed_texts(texts: list) -> list:
    """
    批量把文字变成向量（语义指纹）
    texts：["第一段话", "第二段话", ...]
    返回：[[0.1, 0.2, ...1024个数], [...], ...] 每段话对应一串数字
    """
    return _embedder.embed_documents(texts)       # langchain 的批量向量化接口


def embed_query(query: str) -> list:
    """
    把"用户的一个问题"变成向量（检索时用）
    问题短，单独一个方法，接口也不同（embed_query）
    """
    return _embedder.embed_query(query)           # 返回一串 1024 个的数字


def get_collection():
    """
    拿到向量库的集合对象（不存在就自动创建）
    PersistentClient：数据存到本地 chroma_db/ 文件夹，程序重启数据还在
    """
    client = chromadb.PersistentClient(path=CHROMA_DIR)   # 打开（或创建）本地向量库
    return client.get_or_create_collection(               # 拿到名为 lingxi_kb 的集合
        name=COLLECTION_NAME,
        metadata={"hnsw:space": "cosine"},                # 用"余弦相似度"衡量向量远近（文本检索标准做法）
    )


def rebuild_vector_store(chunks: list):
    """
    重建向量库：把所有文档片段连同向量一起存进去
    chunks：[{"id":"xxx","text":"段落内容","source":"文件名.md"}, ...]
    每次重建都清空旧数据（简单粗暴，语料少时重建只要几秒）
    """
    client = chromadb.PersistentClient(path=CHROMA_DIR)   # 打开本地向量库
    try:                                                  # 尝试删除旧集合
        client.delete_collection(COLLECTION_NAME)         # 有旧的就删掉（保证干净重建）
    except Exception:                                     # 第一次跑没有旧集合会报错
        pass                                              # 报错忽略即可（本来就没有）

    col = client.get_or_create_collection(                # 重新创建空集合
        name=COLLECTION_NAME,
        metadata={"hnsw:space": "cosine"},                # 同样用余弦相似度
    )

    texts = [c["text"] for c in chunks]                   # 抽出所有段落文字
    vectors = embed_texts(texts)                          # 一次性批量算语义指纹（省调用次数）
    col.add(                                              # 存入向量库
        ids=[c["id"] for c in chunks],                    # 每段的唯一编号
        documents=texts,                                  # 每段的原文（检索到时原样返回）
        embeddings=vectors,                               # 每段的语义指纹
        metadatas=[{"source": c["source"]} for c in chunks],  # 每段的来源文件（溯源用）
    )
    print(f"向量库重建完成：{len(chunks)} 个片段已入库 → {CHROMA_DIR}")


def vector_search(query: str, top_k: int = 4) -> list:
    """
    语义检索：拿用户问题，找出语义最相近的 top_k 个片段
    返回：[{"text":段落内容, "source":来源文件, "score":相似度0~1}, ...] 按相似度从高到低
    """
    col = get_collection()                                # 打开向量库
    qv = embed_query(query)                               # 把问题变成向量
    res = col.query(query_embeddings=[qv], n_results=top_k)   # 向量库内部做相似度排序
    out = []                                              # 存整理后的结果
    for doc, meta, dist in zip(                           # zip 同时遍历三组返回值
        res["documents"][0],                              # 命中的段落文字
        res["metadatas"][0],                              # 段落来源
        res["distances"][0],                              # 距离（越小越相似）
    ):
        out.append({                                      # 整理成统一格式
            "text": doc,                                  # 段落原文
            "source": meta.get("source", "未知"),          # 来源文件
            "score": round(1 - dist, 4),                  # 距离转相似度（余弦：相似度=1-距离）
        })
    return out                                            # 交给上层（混合检索/问答生成）
```

## 3. `src/rag/hybrid_retriever.py`（新建）

```python
# src/rag/hybrid_retriever.py
"""
混合检索模块（RAG 的"搜索队长"）
====================================================================
同时指挥两路检索：
  ① 向量路（语义匹配，vector_store.py 负责）
  ② BM25 路（关键词匹配，本文件用 jieba 分词 + rank_bm25 实现）
两路结果用 RRF 算法融合成一个最终排名。

RRF 是什么（大白话）：每路检索都给结果排了个名次，
RRF 说"别比分数了（两边分数没法比），比名次"——
一个片段在两路里名次都靠前，融合分就高。公式：融合分 = Σ 1/(60+名次)。
"""

import json                                        # 读写片段文件（jsonl 格式）
import jieba                                       # 中文分词：把句子切成词语（BM25 只认词）
from rank_bm25 import BM25Okapi                    # BM25 算法：经典的"关键词打分"检索算法
from src.rag.vector_store import vector_search     # 复用向量检索

# 语料片段的存放文件：建库时写入，检索时读出
CHUNKS_FILE = "chroma_db/chunks.jsonl"


def load_chunks() -> list:
    """从 jsonl 文件读出所有片段（BM25 索引要用）"""
    chunks = []                                    # 存读出来的片段
    with open(CHUNKS_FILE, "r", encoding="utf-8") as f:  # 打开片段文件
        for line in f:                             # 逐行读（jsonl 每行是一个 JSON）
            if line.strip():                       # 跳过空行
                chunks.append(json.loads(line))    # 反序列化成 dict 加进列表
    return chunks                                  # 交出全部片段


def bm25_search(query: str, top_k: int = 4) -> list:
    """
    BM25 关键词检索：谁包含的"查询词"多且稀少，谁得分高
    返回：[{"text":..., "source":..., "score":...}, ...] 按得分从高到低
    """
    chunks = load_chunks()                         # 读出全部片段
    tokenized = [list(jieba.cut_for_search(c["text"])) for c in chunks]  # 每个片段切成词列表
    bm25 = BM25Okapi(tokenized)                    # 构建 BM25 索引（语料小，现场建不费时）
    scores = bm25.get_scores(list(jieba.cut_for_search(query)))  # 给每个片段打分
    paired = sorted(zip(chunks, scores), key=lambda x: x[1], reverse=True)  # 按分数降序配对排序
    return [{                                      # 整理成和向量检索相同的输出格式
        "text": c["text"],                         # 片段原文
        "source": c["source"],                     # 来源文件
        "score": round(s, 4),                      # BM25 分数（仅用于本路排名）
    } for c, s in paired[:top_k] if s > 0]         # 只取前 top_k 个、过滤掉 0 分的


def rrf_merge(list_a: list, list_b: list, k: int = 60) -> list:
    """
    RRF 融合：把两路检索的结果按"名次"合成一个排名
    list_a/list_b：两路各自排好序的结果列表；k：RRF 平滑常数（论文标准值60）
    返回：融合后的统一列表（每项多一个 rrf_score 字段）
    """
    pool = {}                                      # 融合池：key=片段原文，value=融合信息
    for rank, item in enumerate(list_a):           # 遍历 A 路结果（rank 从 0 开始）
        pool.setdefault(item["text"], {"item": item, "score": 0})  # 没见过的片段放进池子
        pool[item["text"]]["score"] += 1 / (k + rank + 1)          # A 路名次贡献融合分
    for rank, item in enumerate(list_b):           # 同样处理 B 路结果
        pool.setdefault(item["text"], {"item": item, "score": 0})  # 两路都命中的会在这里累加
        pool[item["text"]]["score"] += 1 / (k + rank + 1)          # B 路名次也贡献融合分
    merged = sorted(pool.values(), key=lambda x: x["score"], reverse=True)  # 按融合分降序
    out = []                                       # 存输出
    for v in merged:                               # 遍历融合结果
        item = dict(v["item"])                     # 复制片段信息
        item["rrf_score"] = round(v["score"], 6)   # 附上融合分（调试/展示用）
        out.append(item)                           # 加进输出
    return out                                     # 交出融合后的统一排名


def hybrid_search(query: str, top_k: int = 4) -> list:
    """
    混合检索总入口（问答子图调用它）：向量路 + BM25 路 → RRF 融合 → 取前 top_k
    返回：[{"text":..., "source":..., "rrf_score":...}, ...]
    """
    vec_results = vector_search(query, top_k)      # 第1路：语义检索
    bm_results = bm25_search(query, top_k)         # 第2路：关键词检索
    merged = rrf_merge(vec_results, bm_results)    # 融合两路排名
    return merged[:top_k]                          # 交出最终的前 top_k 个片段
```

## 4. `src/data/build_kb.py`（新建：建库脚本，手动运行一次）

```python
# src/data/build_kb.py
"""
build_kb.py —— 知识库建库脚本（RAG 的"图书管理员"）
====================================================================
干什么：把 seed_data/kb/ 下的制度文档读出来 → 切成小片段 →
       ① 算向量存进 ChromaDB（语义检索用） ② 写成 jsonl（BM25 检索用）
手动运行一次即可：python -m src.data.build_kb
文档改了就重跑一遍（重建式更新，语料小所以很快）
"""

import json                                        # 写 jsonl 片段文件用
from pathlib import Path                           # 跨平台路径处理
from src.rag.vector_store import rebuild_vector_store, embed_texts  # 向量库写入
from src.rag.hybrid_retriever import CHUNKS_FILE   # 片段文件路径（和检索端共用同一个）

# 语料目录：seed_data/kb/ 下的所有 .md 文件
KB_DIR = Path(__file__).parent.parent.parent / "seed_data" / "kb"
# 每个片段的最大字数：太长的段落检索不精准，超长要再切
MAX_CHUNK_CHARS = 400


def split_doc(text: str) -> list:
    """
    把一篇文档切成检索片段
    切法（简单有效）：先按空行切段；超过400字的段再按句号切成两半
    """
    paras = [p.strip() for p in text.split("\n\n") if p.strip()]  # 按空行切段、去空白
    chunks = []                                   # 存最终片段
    for p in paras:                               # 遍历每个段落
        if len(p) <= MAX_CHUNK_CHARS:             # 段落不长，整段就是一个片段
            chunks.append(p)
        else:                                     # 段落超长：按句号二分（粗但够用）
            mid = len(p) // 2                     # 找中间位置
            cut = p.find("。", mid)               # 从中间往后找第一个句号
            cut = cut + 1 if cut != -1 else len(p)  # 找到就在句号后切，找不到就硬切
            chunks.append(p[:cut])                # 前半段
            chunks.append(p[cut:])                # 后半段
    return [c for c in chunks if len(c) >= 20]    # 太碎的（<20字）不要，检索出来没意义


def run():
    """建库主流程：读文档 → 切片 → 编号 → 入向量库 → 写 jsonl"""
    chunks = []                                   # 存所有片段
    for md_file in sorted(KB_DIR.glob("*.md")):   # 遍历 kb 目录下每个制度文档
        text = md_file.read_text(encoding="utf-8")  # 读出全文
        pieces = split_doc(text)                  # 切成片段
        for i, piece in enumerate(pieces):        # 给每个片段编号
            chunks.append({                       # 片段的三个要素：编号/内容/来源
                "id": f"{md_file.stem}-{i+1:02d}",   # 编号如"采购预算管理办法-03"
                "text": piece,                    # 片段内容
                "source": md_file.name,           # 来源文件名（溯源卡片显示用）
            })
        print(f"已切分 {md_file.name}：{len(pieces)} 个片段")

    if not chunks:                                # 一个片段都没有说明语料目录是空的
        print("警告：seed_data/kb/ 下没有找到语料！")
        return                                    # 直接结束

    rebuild_vector_store(chunks)                  # ① 向量入库（内部会调 embed_texts 算指纹）

    with open(CHUNKS_FILE, "w", encoding="utf-8") as f:  # ② 同一批片段写成 jsonl
        for c in chunks:                          # 逐行写（jsonl 格式：每行一个 JSON）
            f.write(json.dumps(c, ensure_ascii=False) + "\n")  # ensure_ascii 保留中文原文
    print(f"BM25 语料写入完成：{CHUNKS_FILE}")


if __name__ == "__main__":
    run()                                         # 直接运行：python -m src.data.build_kb
```
